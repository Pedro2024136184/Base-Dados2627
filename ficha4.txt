SELECT TO_CHAR(L.DATA_EDICAO, 'DD Mon YYYY') AS DATA_EDICAO, -- Corrigido: Usando 'DD Mon YYYY' para mostrar as 3 letras do mês
       L.Titulo as "Titulo", 
       A.NOME, 
       L.PRECO_TABELA AS "Preço"
FROM LIVROS L
JOIN AUTORES A ON L.CODIGO_AUTOR = A.CODIGO_AUTOR
WHERE L.DATA_EDICAO >= (SYSDATE - 80)
ORDER BY L.DATA_EDICAO;
EXEC SQLCHECK('FDWLWHTEXWQAIVG');


SELECT DISTINCT L.TITULO AS TITULO, 
       L.PRECO_TABELA AS PRECO_TABELA -- Corrigido para PRECO_TABELA (sem aspas/espaço)
FROM LIVROS L
JOIN CONTEM C ON L.CODIGO_LIVRO = C.CODIGO_LIVRO
JOIN VENDAS V ON C.CODIGO_VENDA = V.CODIGO_VENDA
JOIN CLIENTES CL ON V.CODIGO_CLIENTE = CL.CODIGO_CLIENTE
WHERE UPPER(L.GENERO) = 'INFORMÁTICA'
  AND CL.MORADA LIKE '%Odivelas%'
  AND V.DATA_VENDA BETWEEN TO_DATE('12-MAY-2022', 'DD-MON-YYYY', 'NLS_DATE_LANGUAGE=ENGLISH') 
                       AND TO_DATE('12-JUN-2024', 'DD-MON-YYYY', 'NLS_DATE_LANGUAGE=ENGLISH')
ORDER BY L.TITULO;
EXEC SQLCHECK('FDKWHQSFWQZCJIY');


SELECT DISTINCT L.TITULO AS TITULO, 
       TO_CHAR(L.DATA_EDICAO, 'YYYY-MM') AS ANO_MES
FROM LIVROS L
JOIN CONTEM C ON L.CODIGO_LIVRO = C.CODIGO_LIVRO
JOIN VENDAS V ON C.CODIGO_VENDA = V.CODIGO_VENDA
WHERE UPPER(L.GENERO) = 'INFORMÁTICA'
  AND TO_CHAR(L.DATA_EDICAO, 'YYYY-MM') = TO_CHAR(V.DATA_VENDA, 'YYYY-MM')
ORDER BY ANO_MES;
EXEC SQLCHECK('FDZHRUYGKCFZKUQ');


SELECT DISTINCT A.NOME AS NOME, 
       A.GENERO_PREFERIDO
FROM AUTORES A
JOIN LIVROS L ON A.CODIGO_AUTOR = L.CODIGO_AUTOR
WHERE L.GENERO = 'Fição'
  AND L.PAGINAS < 100
  AND EXTRACT(YEAR FROM L.DATA_EDICAO) = 2022
  AND (
    (L.DATA_EDICAO BETWEEN DATE '2022-06-21' AND DATE '2022-09-21')
    OR (L.DATA_EDICAO BETWEEN DATE '2022-09-22' AND DATE '2022-12-21')
  )
ORDER BY A.NOME;
EXEC SQLCHECK('FDYDDOXHJVAQLTT');

SELECT 
    L.Genero,
    L.Titulo AS "Titulo",
    TO_CHAR(L.Data_Edicao, 'DD-MM-YYYY') AS Data_Edicao,
    TRUNC(MONTHS_BETWEEN(SYSDATE, L.Data_Edicao) / 12) AS "Num. de anos"
FROM 
    LIVROS L
    JOIN EDITORAS E ON L.Codigo_Editora = E.Codigo_Editora
    JOIN AUTORES A ON L.Codigo_Autor = A.Codigo_Autor
WHERE 
    UPPER(E.Nome) = 'PORTO EDITORA'
    AND UPPER(L.Genero) IN ('ROMANCE', 'POLICIAL', 'AVENTURA')
    AND UPPER(L.Genero) = UPPER(A.Genero_Preferido)
ORDER BY 
    L.Genero,
    L.Titulo;

EXEC SQLCHECK('FDSHQBGIDGMMMKI');

SELECT 
    a.nome AS NOME,
    TRUNC(MONTHS_BETWEEN(SYSDATE, a.data_nascimento) / 12) AS IDADE
FROM autores a
WHERE NOT EXISTS (
    SELECT 1
    FROM livros l
    WHERE l.codigo_autor = a.codigo_autor
)
ORDER BY a.nome;
EXEC SQLCHECK('FDCZJEHKLJPCOWU');
