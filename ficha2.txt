SELECT Titulo   AS TITULO,
       Genero   AS GENERO,
       Preco_Tabela AS PRECO
FROM LIVROS
WHERE Preco_Tabela >= 30
  AND Preco_Tabela <  40
ORDER BY Preco_Tabela, Titulo;

SELECT titulo, genero FROM Livros WHERE (titulo LIKE 'O%' AND titulo LIKE '%st%') OR (titulo LIKE 'R%' AND titulo LIKE '%eng%') ORDER BY preco_tabela;

SELECT titulo, genero, preco_tabela FROM Livros WHERE titulo LIKE 'A%' AND NOT (genero = 'Policial' OR genero = 'Romance' OR genero = 'Drama') ORDER BY titulo ASC;

EXEC SQLCHECK('FBVLIMIOEUBZQBM');
