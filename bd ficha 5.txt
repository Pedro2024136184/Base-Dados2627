SELECT
    L.TITULO,
    MAX(CON.PRECO_UNITARIO) AS "P_MAIS_ALTO",
    MIN(CON.PRECO_UNITARIO) AS "P_MAIS_BAIXO",
    ROUND(AVG(CON.PRECO_UNITARIO), 1) AS "P_MEDIO"
FROM
    LIVROS L
JOIN
    CONTEM CON ON CON.CODIGO_LIVRO = L.CODIGO_LIVRO
WHERE
    UPPER(L.GENERO) = 'INFORMÁTICA'
    AND EXTRACT(MONTH FROM L.DATA_EDICAO) = 1
    AND EXTRACT(YEAR FROM L.DATA_EDICAO) = 2024
GROUP BY
    L.TITULO
ORDER BY
    "P_MAIS_ALTO",
    L.TITULO;

SELECT
    L.TITULO,
    L.PRECO_TABELA AS "PRECO",
    SUM(CON.QUANTIDADE) AS "NUM_VENDIDOS",
    ROUND(SUM(CON.QUANTIDADE * L.PRECO_TABELA), 2) AS "REC_ESPERADA",
    ROUND(SUM(CON.QUANTIDADE * CON.PRECO_UNITARIO), 2) AS "REC_EFECTIVA"
FROM 
    LIVROS L
JOIN 
    CONTEM CON ON CON.CODIGO_LIVRO = L.CODIGO_LIVRO
JOIN 
    VENDAS V ON V.CODIGO_VENDA = CON.CODIGO_VENDA
WHERE 
    L.PRECO_TABELA > 80
    AND EXTRACT(YEAR FROM V.DATA_VENDA) = 2024
GROUP BY 
    L.TITULO, L.PRECO_TABELA
ORDER BY 
    L.TITULO;

SELECT
    C.NOME,
    SUM(CON.QUANTIDADE) AS "N.Livros",
    ROUND(AVG(CON.PRECO_UNITARIO), 2) AS "Preco Medio",
    COUNT(DISTINCT L.CODIGO_AUTOR) AS "N.Autor Diferent"
FROM
    CLIENTES C
JOIN
    VENDAS V ON V.CODIGO_CLIENTE = C.CODIGO_CLIENTE
JOIN
    CONTEM CON ON CON.CODIGO_VENDA = V.CODIGO_VENDA
JOIN
    LIVROS L ON L.CODIGO_LIVRO = CON.CODIGO_LIVRO
JOIN
    EDITORAS E ON E.CODIGO_EDITORA = L.CODIGO_EDITORA
WHERE
    UPPER(C.MORADA) LIKE '%FIGUEIRA DA FOZ%'
GROUP BY
    C.NOME
HAVING
    SUM(CON.QUANTIDADE) <= 50
    AND AVG(CON.PRECO_UNITARIO) <= 11
    AND COUNT(DISTINCT L.CODIGO_EDITORA) > 1
ORDER BY
    C.NOME;

SELECT 
    A.NOME,
    CASE 
        WHEN COUNT(L.CODIGO_LIVRO) > 0 THEN  
            'Escreveu ' || COUNT(L.CODIGO_LIVRO) || 
            ' livros de ' || L.GENERO
        ELSE
            'Não escreveu Livros'
    END AS "Num.de Livros que escreveu"
FROM
    AUTORES A
LEFT JOIN
    LIVROS L ON L.CODIGO_AUTOR = A.CODIGO_AUTOR
WHERE
    TRUNC(MONTHS_BETWEEN(SYSDATE, A.DATA_NASCIMENTO) / 12) > 50
    AND UPPER(A.GENERO_PREFERIDO) = 'ROMANCE'
GROUP BY
    A.NOME, 
    L.GENERO
ORDER BY
    A.NOME;

SELECT
    LJ.NOME AS "NOME_LOJA",
    SUM(CON.QUANTIDADE) AS "QUANTIDADE_VENDIDA",
    ROUND(AVG(CON.PRECO_UNITARIO), 2) AS "PRECO_MEDIO"
FROM
    LOJAS LJ
JOIN
    VENDAS V ON V.CODIGO_LOJA = LJ.CODIGO_LOJA
JOIN
    CONTEM CON ON CON.CODIGO_VENDA = V.CODIGO_VENDA
JOIN
    LIVROS L ON L.CODIGO_LIVRO = CON.CODIGO_LIVRO
WHERE
    UPPER(L.GENERO) = 'POLICIAL'
    AND EXTRACT(YEAR FROM V.DATA_VENDA) = EXTRACT(YEAR FROM SYSDATE)
GROUP BY
    LJ.NOME
HAVING
    SUM(CON.QUANTIDADE) >= 20   
ORDER BY
    "QUANTIDADE_VENDIDA" DESC,
    LJ.NOME ;

         TO_CHAR(DATA_TRANSFERENCIA, 'MONTH')
ORDER BY TO_CHAR(DATA_TRANSFERENCIA, 'MONTH'), CEIL(EXTRACT(DAY FROM DATA_TRANSFERENCIA)/7);

SELECT
    C.NOME,
    SUM(CON.QUANTIDADE) AS "QUANTIDADE_VENDIDA",
    COUNT(DISTINCT V.CODIGO_LOJA) AS "NUM_LOJAS"
FROM
    CLIENTES C
JOIN
    VENDAS V ON V.CODIGO_CLIENTE = C.CODIGO_CLIENTE
JOIN
    CONTEM CON ON CON.CODIGO_VENDA = V.CODIGO_VENDA
JOIN
    LIVROS L ON L.CODIGO_LIVRO = CON.CODIGO_LIVRO
WHERE
    UPPER(C.MORADA) LIKE '%BARCELOS%'
    AND UPPER(L.GENERO) = 'POLICIAL'
GROUP BY
    C.NOME
ORDER BY
    C.NOME;
