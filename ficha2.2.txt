SELECT Titulo   AS TITULO,
       Genero   AS GENERO,
       Preco_Tabela AS PRECO
FROM LIVROS
WHERE Preco_Tabela >= 30
  AND Preco_Tabela <  40
ORDER BY Preco_Tabela, Titulo;

where A.morada like '%Portalegre%' And L.Genero='Aventura'

SELECT DISTINCT A.Nome, TO_CHAR(A.Data_Nascimento, 'YYYY') AS Ano_de_Nascimento
FROM AUTORES A
JOIN LIVROS L ON A.Codigo_Autor = L.Codigo_Autor
WHERE A.Genero_Preferido = L.Genero
ORDER BY Ano_de_Nascimento, A.Nome;
EXEC SQLCHECK('FCZNODMEIXEJHZJ');

SELECT DISTINCT
    L.titulo,
    L.genero,
    L.preco_tabela
FROM
    Livros L
JOIN
    Contem CO ON L.codigo_livro = CO.codigo_livro  -- Liga Livros à tabela de associação
JOIN
    Vendas V ON CO.codigo_venda = V.codigo_venda  -- Liga a tabela de associação à Venda
JOIN
    Clientes C ON V.codigo_cliente = C.codigo_cliente  -- Liga a Venda ao Cliente
WHERE
    C.morada LIKE '%Lisboa%'
ORDER BY
    L.titulo ASC;
    EXEC SQLCHECK('FCRBLZVFBDUWIVB');
